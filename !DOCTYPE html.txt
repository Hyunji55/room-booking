    function escapeHTML(str) {
      if (!str) return "";
      return String(str)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }

    function showToast(message, type = "success") {
      const toast = document.getElementById("toast");
      toast.textContent = message;
      toast.className = `show ${type}`;
      setTimeout(() => {
        toast.className = "";
      }, 3000);
    }

    function getLocalTodayString() {
      const d = new Date();
      const year = d.getFullYear();
      const month = String(d.getMonth() + 1).padStart(2, '0');
      const day = String(d.getDate()).padStart(2, '0');
      return `${year}-${month}-${day}`;
    }

    function getMonday(d) {
      d = new Date(d);
      const day = d.getDay();
      const diff = d.getDate() - day + (day === 0 ? -6 : 1); // 월요일 기준
      return new Date(d.setDate(diff));
    }

    function formatDateStr(dateObj) {
      const y = dateObj.getFullYear();
      const m = String(dateObj.getMonth() + 1).padStart(2, '0');
      const d = String(dateObj.getDate()).padStart(2, '0');
      return `${y}-${m}-${d}`;
    }

    function formatDisplayDate(dateStr) {
      const [y, m, d] = dateStr.split('-');
      const dateObj = new Date(y, m - 1, d);
      const days = ['일', '월', '화', '수', '목', '금', '토'];
      const dayName = days[dateObj.getDay()];
      return `${parseInt(m)}월 ${parseInt(d)}일 (${dayName})`;
    }

    function initDateInputs() {
      const todayStr = getLocalTodayString();
      const todayDate = new Date();
      // 주말이면 다음 월요일 기본값
      if (todayDate.getDay() === 0 || todayDate.getDay() === 6) {
        const nextMon = new Date(currentMonday);
        nextMon.setDate(nextMon.getDate() + 7);
        document.getElementById("reserved-date").value = formatDateStr(nextMon);
      } else {
        document.getElementById("reserved-date").value = todayStr;
      }
      document.getElementById("reserved-date").min = todayStr;
    }

    async function fetchRooms() {

      searchRoomSelect.innerHTML = '<option value="">모든 특별실</option>';

    async function handleReservationSubmit(event) {
      event.preventDefault();

      const roomId = document.getElementById("room-select").value;
      const reservedDate = document.getElementById("reserved-date").value;
      const period = parseInt(document.getElementById("period-select").value);
      const teacherName = document.getElementById("teacher-name").value.trim();
      const grade = parseInt(document.getElementById("grade-select").value);
      const classNoVal = document.getElementById("class-no").value;
      const classNo = classNoVal ? parseInt(classNoVal) : null;
      const purpose = document.getElementById("purpose").value.trim();

      // 프론트엔드 검사
      if (!reservedDate || !teacherName) {
        showToast("필수 입력값을 확인해주세요.", "error");
        return;
      }
      const selectedDateObj = new Date(reservedDate);
      const todayObj = new Date(getLocalTodayString());
      if (selectedDateObj < todayObj) {
        showToast("지난 날짜는 예약할 수 없습니다.", "error");
        return;
      }
      const dayOfWeek = selectedDateObj.getDay();
      if (dayOfWeek === 0 || dayOfWeek === 6) {
        showToast("주말에는 예약할 수 없습니다.", "error");
        return;
      }

      const submitBtn = document.getElementById("submit-btn");
      submitBtn.disabled = true;
      submitBtn.textContent = "저장 중…";

      const payload = {
        room_id: parseInt(roomId),
        reserved_date: reservedDate,
        period: period,
        teacher_name: teacherName,
        grade: grade,
        class_no: classNo,
        purpose: purpose || null
      };

      const { data, error } = await db.from("reservations").insert(payload);

      submitBtn.disabled = false;
      submitBtn.textContent = "예약하기";

      if (error) {
        handleDbError(error);
        if (error.code === '23505') {
          loadWeeklySchedule(); // 다른 사람이 먼저 예약한 경우 시간표 동기화
        }
        return;
      }

      const roomObj = roomsData.find(r => r.id == roomId);
      const roomName = roomObj ? roomObj.name : "특별실";
      showToast(`${formatDisplayDate(reservedDate)} ${period}교시 ${roomName} 예약 완료!`, "success");
      
      document.getElementById("purpose").value = "";
      
      // 시간표, 목록, 통계 새로고침 및 해당 특별실로 시간표 맞춤
      document.getElementById("schedule-room-select").value = roomId;
      loadWeeklySchedule();
      loadReservationList();
    }

    async function loadWeeklySchedule() {
      const roomId = document.getElementById("schedule-room-select").value;
      if (!roomId) return;

      // 월~금 날짜 계산
      const weekDates = [];
      let tempDate = new Date(currentMonday);
      for (let i = 0; i < 5; i++) {
        weekDates.push(formatDateStr(tempDate));
        tempDate.setDate(tempDate.getDate() + 1);
      }

      // 기간 표시 갱신
      document.getElementById("week-range-display").textContent = 
        `${formatDisplayDate(weekDates[0])} ~ ${formatDisplayDate(weekDates[4])}`;

      // 해당 주의 예약 데이터 조회
      const { data: reservations, error } = await db
        .from("reservations")
        .select("*")
        .eq("room_id", roomId)
        .gte("reserved_date", weekDates[0])
        .lte("reserved_date", weekDates[4]);

      if (error) {
        showToast("시간표를 불러오지 못했습니다.", "error");
        return;
      }

      // 맵핑 생성 (reserved_date_period)
      const resMap = {};
      (reservations || []).forEach(r => {
        resMap[`${r.reserved_date}_${r.period}`] = r;
      });

      const todayStr = getLocalTodayString();
      const tbody = document.getElementById("schedule-tbody");
      tbody.innerHTML = "";

      for (let p = 1; p <= 6; p++) {
        let tr = document.createElement("tr");
        let th = document.createElement("th");
        th.textContent = `${p}교시`;
        tr.appendChild(th);

        for (let d = 0; d < 5; d++) {
          const dateStr = weekDates[d];
          const cellKey = `${dateStr}_${p}`;
          const res = resMap[cellKey];
          const td = document.createElement("td");

          const isPast = dateStr < todayStr;
          const isToday = dateStr === todayStr;

          if (isPast) td.classList.add("past");
          if (isToday) td.classList.add("today");

          if (res) {
            td.classList.add("slot-reserved");
            const classStr = res.class_no ? ` ${res.grade}학년 ${res.class_no}반` : ` ${res.grade}학년`;
            td.innerHTML = `<strong>${escapeHTML(res.teacher_name)}</strong><br><span style="font-size:12px;">${classStr}</span>`;
            if (res.purpose) {
              td.title = `사용 목적: ${res.purpose}`;
            }
          } else {
            if (isPast) {
              td.textContent = "-";
            } else {
              td.textContent = "+ 예약";
              td.style.cursor = "pointer";
              td.style.color = "var(--text-sub)";
              td.onclick = () => {
                document.getElementById("room-select").value = roomId;
                document.getElementById("reserved-date").value = dateStr;
                document.getElementById("period-select").value = p;
                document.getElementById("teacher-name").focus();
                window.scrollTo({ top: 0, behavior: 'smooth' });
              };
            }
          }
          tr.appendChild(td);
        }
        tbody.appendChild(tr);
      }
    }

    async function loadReservationList() {
      const teacherQuery = document.getElementById("search-teacher").value.trim();
      const roomQuery = document.getElementById("search-room").value;
      const dateQuery = document.getElementById("search-date").value;
      const showPast = document.getElementById("search-past").checked;
      const todayStr = getLocalTodayString();

      let query = db.from("reservations").select("*, rooms(name)").order("reserved_date", { ascending: true }).order("period", { ascending: true }).limit(200);

      if (!showPast) {
        query = query.gte("reserved_date", todayStr);
      }
      if (roomQuery) {
        query = query.eq("room_id", roomQuery);
      }
      if (dateQuery) {
        query = query.eq("reserved_date", dateQuery);
      }
      if (teacherQuery) {
        query = query.ilike("teacher_name", `%${teacherQuery}%`);
      }

      const { data, error } = await query;
      const tbody = document.getElementById("reservation-list-tbody");
      tbody.innerHTML = "";

      if (error) {
        showToast("검색 중 오류가 발생했습니다.", "error");
        return;
      }

      if (!data || data.length === 0) {
        tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; color:var(--text-sub); padding:24px;">조건에 맞는 예약이 없습니다.</td></tr>`;
        return;
      }

      data.forEach(res => {
        const tr = document.createElement("tr");
        const roomName = res.rooms ? res.rooms.name : "특별실";
        const classStr = res.class_no ? `${res.grade}학년 ${res.class_no}반` : `${res.grade}학년`;

        tr.innerHTML = `
          <td>${formatDisplayDate(res.reserved_date)}</td>
          <td><strong>${res.period}교시</strong> · ${escapeHTML(roomName)}</td>
          <td>${escapeHTML(res.teacher_name)} 선생님 · ${classStr}</td>
          <td style="color:var(--text-sub); max-width:200px; white-space:nowrap; overflow:hidden; text-overflow:ellipsis;" title="${escapeHTML(res.purpose || '')}">${escapeHTML(res.purpose || '-')}</td>
          <td><button class="cancel-btn" onclick="cancelReservation(${res.id}, '${res.reserved_date}', ${res.period}, '${escapeHTML(roomName)}', '${escapeHTML(res.teacher_name)}')">취소</button></td>
        `;
        tbody.appendChild(tr);
      });
    }

    async function cancelReservation(id, dateStr, period, roomName, teacherName) {
      if (!confirm(`${formatDisplayDate(dateStr)} ${period}교시 ${roomName} (${teacherName}) 예약을 취소할까요?`)) {
        return;
      }

      const { data, error } = await db.from("reservations").delete().eq("id", id).select();

      if (error) {
        handleDbError(error);
        return;
      }

      if (!data || data.length === 0) {
        showToast("이미 취소되었거나 삭제 권한이 없습니다.", "error");
      } else {
        showToast("예약이 취소되었습니다.", "success");
      }

      loadWeeklySchedule();
      loadReservationList();
    }

    async function loadStats() {
      const periodType = document.getElementById("stats-period-type").value;
      const today = new Date();
      let startDate, endDate, weekdaysCount = 0;

      if (periodType === 'week') {
        const mon = getMonday(today);
        const fri = new Date(mon);
        fri.setDate(fri.getDate() + 4);
        startDate = formatDateStr(mon);
        endDate = formatDateStr(fri);
        weekdaysCount = 5;
      } else {
        const y = today.getFullYear();
        const m = today.getMonth();
        const firstDay = new Date(y, m, 1);
        const lastDay = new Date(y, m + 1, 0);
        startDate = formatDateStr(firstDay);
        endDate = formatDateStr(lastDay);

        // 이번 달 평일 수 계산
        let cur = new Date(firstDay);
        while (cur <= lastDay) {
          const day = cur.getDay();
          if (day !== 0 && day !== 6) weekdaysCount++;
          cur.setDate(cur.getDate() + 1);
        }
      }

      const totalSlotsPerRoom = weekdaysCount * 6;

      // 기간 내 예약 데이터 불러오기
      const { data: reservations, error } = await db
        .from("reservations")
        .select("*, rooms(name)")
        .gte("reserved_date", startDate)
        .lte("reserved_date", endDate);

      if (error) {
        showToast("통계 데이터를 불러오지 못했습니다.", "error");
        return;
      }

      const resList = reservations || [];
      const totalReservations = resList.length;

      // 요약 줄 렌더링
      const summaryBox = document.getElementById("stats-summary-box");
      summaryBox.innerHTML = `
        <div class="stat-box"><div>조회 기간</div><div class="value" style="font-size:16px; margin-top:6px;">${startDate} ~ ${endDate}</div></div>
        <div class="stat-box"><div>평일 수</div><div class="value">${weekdaysCount}일</div></div>
        <div class="stat-box"><div>특별실당 총 칸수</div><div class="value">${totalSlotsPerRoom}칸</div></div>
        <div class="stat-box"><div>총 예약 건수</div><div class="value">${totalReservations}건</div></div>
      `;

      // 1. 특별실별 이용률 계산
      const roomUsageMap = {};
      roomsData.forEach(r => { roomUsageMap[r.id] = { name: r.name, count: 0 }; });
      resList.forEach(r => {
        if (roomUsageMap[r.room_id]) roomUsageMap[r.room_id].count++;
      });

      const roomNames = [];
      const roomUsageRates = [];
      const roomCounts = [];

      Object.values(roomUsageMap).forEach(item => {
        roomNames.push(item.name);
        const rate = totalSlotsPerRoom > 0 ? ((item.count / totalSlotsPerRoom) * 100).toFixed(1) : 0;
        roomUsageRates.push(parseFloat(rate));
        roomCounts.push(item.count);
      });

      renderRoomUsageChart(roomNames, roomUsageRates, roomCounts, totalSlotsPerRoom);

      // 2. 요일·교시별 혼잡도 계산 (월~금: 1~5, 교시: 1~6)
      // 버블 데이터 구조: { x: 요일(1~5), y: 교시(1~6), r: 건수 }
      const congestionMap = {};
      for (let d = 1; d <= 5; d++) {
        for (let p = 1; p <= 6; p++) {
          congestionMap[`${d}_${p}`] = 0;
        }
      }
      resList.forEach(r => {
        const dObj = new Date(r.reserved_date);
        const dayOfWeek = dObj.getDay(); // 1(월)~5(금)
        if (dayOfWeek >= 1 && dayOfWeek <= 5) {
          const key = `${dayOfWeek}_${r.period}`;
          if (congestionMap[key] !== undefined) congestionMap[key]++;
        }
      });

      const bubbleData = [];
      for (let d = 1; d <= 5; d++) {
        for (let p = 1; p <= 6; p++) {
          const count = congestionMap[`${d}_${p}`];
          if (count > 0) {
            bubbleData.push({ x: d, y: p, r: Math.max(count * 4, 6) }); // 시각적 크기 조정
          }
        }
      }
      renderCongestionChart(bubbleData);

      // 3. 학년별 이용 횟수 계산 (1~6학년)
      const gradeCounts = [0, 0, 0, 0, 0, 0];
      resList.forEach(r => {
        if (r.grade >= 1 && r.grade <= 6) {
          gradeCounts[r.grade - 1]++;
        }
      });
      renderGradeCountChart(gradeCounts);
    }

    function renderRoomUsageChart(labels, rates, counts, totalSlots) {
      const ctx = document.getElementById("chart-room-usage").getContext("2d");
      if (chartRoomUsageInstance) chartRoomUsageInstance.destroy();

      chartRoomUsageInstance = new Chart(ctx, {
        type: 'bar',
        data: {
          labels: labels,
          datasets: [{
            label: '이용률 (%)',
            data: rates,
            backgroundColor: '#10b981',
            borderRadius: 6
          }]
        },
        options: {
          indexAxis: 'y',
          responsive: true,
          maintainAspectRatio: false,
          scales: {
            x: { min: 0, max: 100, ticks: { callback: v => v + '%' } }
          },
          plugins: {
            tooltip: {
              callbacks: {
                label: function(context) {
                  const idx = context.dataIndex;
                  return ` ${context.raw}% (${counts[idx]}건 / ${totalSlots}칸)`;
                }
              }
            }
          }
        }
      });
    }

    function renderCongestionChart(bubbleData) {
      const ctx = document.getElementById("chart-congestion").getContext("2d");
      if (chartCongestionInstance) chartCongestionInstance.destroy();

      const dayLabels = ['', '월', '화', '수', '목', '금'];

      chartCongestionInstance = new Chart(ctx, {
        type: 'bubble',
        data: {
          datasets: [{
            label: '예약 건수 혼잡도',
            data: bubbleData,
            backgroundColor: 'rgba(16, 185, 129, 0.6)'
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          scales: {
            x: {
              min: 0.5,
              max: 5.5,
              ticks: {
                stepSize: 1,
                callback: v => dayLabels[v] || ''
              }
            },
            y: {
              min: 0.5,
              max: 6.5,
              reverse: true, // 1교시가 위로 오도록
              ticks: {
                stepSize: 1,
                callback: v => `${v}교시`
              }
            }
          }
        }
      });
    }

    // 오류 처리 매핑
    // ==========================================
    function handleDbError(error) {
      const code = error.code;
      const message = error.message;

      if (code === '23505') {
        showToast("이미 예약된 시간입니다. 다른 교시나 날짜를 골라 주세요.", "error");
      } else if (code === '23514') {
        showToast("입력값이 허용 범위를 벗어났습니다.", "error");
      } else if (code === '23503') {
        showToast("존재하지 않는 특별실입니다. 새로고침 후 다시 시도해 주세요.", "error");
      } else if (code === '42501') {
        showToast("권한이 없습니다. Supabase의 RLS 정책을 확인해 주세요.", "error");
      } else {
        showToast(`오류가 발생했습니다: ${message}`, "error");
      }
    }
